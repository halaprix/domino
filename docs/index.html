for (const vault of vaults) {
  // Call 1: get user's vault shares
  const shares = await vault.balanceOf(user);

  // Call 2: convert to assets (depends on Call 1)
  const assets = await vault.convertToAssets(shares);

  // Result: N×M RPC calls (slow)
}
check THE SOLUTION 
// M multicalls — results from step 1 flow to step 2
const executor = new Eip1193Executor(provider);
const results = await runMultistepTasks(executor, [
  defineTask((t) => {
    const balance = t.call({...});
    const assets = t.call({ args: [balance] });
    return { balance, assets };
  })
]);

Quick Start

import { createPublicClient, http } from "viem"
import { Eip1193Executor, resolveErc4626Vault } from "@halaprix/domino"

const provider = createPublicClient({ transport: http() });
const executor = new Eip1193Executor(provider);

const vault = await resolveErc4626Vault({
  executor,
  vault: "0x...",
  owner: "0x...",
});
